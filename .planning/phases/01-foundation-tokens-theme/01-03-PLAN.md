    ```tsx
    // src/components/theme-toggle.tsx
    "use client";

    import { useEffect, useState } from "react";
    import { useTheme } from "next-themes";

    export function ThemeToggle() {
      const { resolvedTheme, setTheme } = useTheme();
      const [mounted, setMounted] = useState(false);

      // Hydration guard: useTheme() returns undefined on the server pass (no localStorage).
      // Render a layout-preserving disabled placeholder with the same 44x44 footprint
      // to prevent CLS and a hydration mismatch warning (RESEARCH §Pitfall 5).
      useEffect(() => setMounted(true), []);
      if (!mounted) {
        return (
          <button
            type="button"
            disabled
            aria-label="Toggle theme"
            className="h-11 w-11"
          />
        );
      }

      const next = resolvedTheme === "dark" ? "light" : "dark";
      return (
        <button
          type="button"
          aria-label={`Switch to ${next} theme`}
          onClick={() => setTheme(next)}
          className="h-11 w-11 inline-flex items-center justify-center text-text-primary"
        >
          {resolvedTheme === "dark" ? "☀" : "☾"}
        </button>
      );
    }
    ```

    Notes (every detail is load-bearing per UI-SPEC §"ThemeToggle affordance"):
    - `<button type="button">` — NEVER `<div>` or `<span>` with `onClick` (PITFALLS §13 + CLAUDE.md forbidden pattern: divs-with-onClick masquerading as buttons).
    - `h-11 w-11` = 44×44 CSS px minimum touch target (A11Y-06).
    - Icons: raw Unicode glyphs `☀` (sun — shown when in dark mode, offering "switch to light") / `☾` (moon — shown when in light mode, offering "switch to dark"). `lucide-react` is NOT installed in Phase 1 per RESEARCH §Supporting deferred deps; Phase 3 swaps to `<Sun />` / `<Moon />`.
    - `aria-label` is dynamic: `` `Switch to ${next} theme` `` — never "Open menu / Close menu" style (PITFALLS §13: screen readers re-announce on every change → noise).
    - Placeholder (unmounted) `<button disabled className="h-11 w-11" />` preserves layout space and keeps the aria-label present. `aria-label="Toggle theme"` on the placeholder is acceptable static fallback.
    - NO hover state, NO background, NO border in Phase 1 — UI-SPEC §Motion Contract forbids motion, §"ThemeToggle affordance" forbids background/border in idle state. Phase 3 adds `hover:bg-surface`.
    - Focus ring inherits from the global `:focus-visible` rule in `globals.css` — no per-component override.
    - `inline-flex items-center justify-center` centers the Unicode glyph inside the 44×44 hit area.
    - `text-text-primary` makes the glyph inherit the themed foreground color (readable in both themes via cascade through the token).

| localStorage `portfolio-theme` → returning-user paint | User preference persisted; no PII; lowest-sensitivity data |
